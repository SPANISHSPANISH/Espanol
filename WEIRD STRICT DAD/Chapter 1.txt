local function GetHRP()
   local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
   return char:FindFirstChild("HumanoidRootPart")
end

local function GetHumanoid()
   local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
   return char:FindFirstChildOfClass("Humanoid")
end

local function pressE()
   VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
   task.wait(0.01)
   VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
end

local function SafeTeleport(pos)
   local hrp = GetHRP()
   if hrp then
      hrp.CFrame = CFrame.new(pos)
   end
end

local function StartFirstPerson()
   if FirstPersonConnection then return end
   FirstPersonConnection = RunService.RenderStepped:Connect(function()
      LocalPlayer.CameraMode = Enum.CameraMode.LockFirstPerson
   end)
end

local function StopFirstPerson()
   if FirstPersonConnection then
      FirstPersonConnection:Disconnect()
      FirstPersonConnection = nil
   end
   LocalPlayer.CameraMode = Enum.CameraMode.Classic
end

local function EnableInstantPrompts()
   for _, v in ipairs(workspace:GetDescendants()) do
      if v:IsA("ProximityPrompt") then
         v.HoldDuration = 0
      end
   end
end

local function FindTool(name)
   local backpack = LocalPlayer:FindFirstChild("Backpack")
   local character = LocalPlayer.Character
   if backpack and backpack:FindFirstChild(name) then return backpack[name] end
   if character and character:FindFirstChild(name) then return character[name] end
   return nil
end

local function EquipTool(name)
   local tool = FindTool(name)
   local hum = GetHumanoid()
   if tool and hum then
      hum:EquipTool(tool)
      return true
   end
   return false
end

local function GetDadFolder()
   return workspace:FindFirstChild("Game") and workspace.Game:FindFirstChild("dad")
end

MainTab:CreateButton({
   Name = "Auto Pickup Trashes",
   Callback = function()
      if FinishedCleaning then
         Rayfield:Notify({Title = "Error", Content = "Already cleaned", Duration = 3})
         return
      end
      task.spawn(function()
         StartFirstPerson()
         EnableInstantPrompts()
         Rayfield:Notify({Title = "Auto Trash", Content = "Started", Duration = 3})

         while true do
            local folder = workspace:FindFirstChild("Game") and workspace.Game:FindFirstChild("trashes") and workspace.Game.trashes:FindFirstChild("P2")
            if not folder then break end
            local parts = {}
            for _, v in ipairs(folder:GetChildren()) do
               if v:IsA("BasePart") then table.insert(parts, v) end
            end
            if #parts == 0 then break end
            local target = parts[math.random(1, #parts)]
            local hrp = GetHRP()
            if hrp and target then
               hrp.CFrame = target.CFrame + Vector3.new(0, 3, 0)
               task.wait(0.22)
               pressE()
               task.wait(0.18)
            end
            task.wait(0.1)
         end

         local trashBin = workspace:FindFirstChild("Game") and workspace.Game:FindFirstChild("TrashBin")
         if trashBin then
            for i = 1, 35 do
               local hrp = GetHRP()
               if hrp then hrp.CFrame = trashBin.CFrame + Vector3.new(0, 3, 0) end
               pressE()
               task.wait(0.25)
            end
         end
         FinishedCleaning = true
         StopFirstPerson()
         Rayfield:Notify({Title = "Auto Trash", Content = "Finished!", Duration = 4})
      end)
   end,
})

InstantTab:CreateToggle({
   Name = "Enable Space from Dad",
   CurrentValue = false,
   Flag = "SpaceToggle",
   Callback = function(Value)
      SpaceFromDadEnabled = Value
      AlreadyTeleportedOnce = false
      if SpaceConnection then SpaceConnection:Disconnect() SpaceConnection = nil end
      if SpaceESP then SpaceESP:Destroy() SpaceESP = nil end

      if Value then
         local hrp = GetHRP()
         if hrp then
            local part = Instance.new("Part")
            part.Name = "DadSpaceESP"
            part.Anchored = true
            part.CanCollide = false
            part.Transparency = SpaceESPVisible and SpaceESPTransparency or 1
            part.Color = Color3.fromRGB(255, 0, 0)
            part.Material = Enum.Material.Neon
            part.Size = Vector3.new(SpaceFromDadValue*2, 1, SpaceFromDadValue*2)
            part.CFrame = hrp.CFrame * CFrame.new(0, -3, 0)
            part.Parent = workspace
            SpaceESP = part
         end

         SpaceConnection = RunService.Heartbeat:Connect(function()
            local hrp = GetHRP()
            local folder = GetDadFolder()
            if not hrp or not folder then return end
            if SpaceESP then
               SpaceESP.CFrame = hrp.CFrame * CFrame.new(0, -3, 0)
               SpaceESP.Size = Vector3.new(SpaceFromDadValue*2, 1, SpaceFromDadValue*2)
            end
            local function check(model)
               if not model then return end
               local root = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
               if root then
                  local dist = (hrp.Position - root.Position).Magnitude
                  if dist <= SpaceFromDadValue and not AlreadyTeleportedOnce then
                     AlreadyTeleportedOnce = true
                     SafeTeleport(EmergencyBedPos)
                     task.wait(0.1)
                     pressE()
                     Rayfield:Notify({Title = "Dad Alert", Content = "Teleported to bed!", Duration = 3})
                  elseif dist > SpaceFromDadValue then
                     AlreadyTeleportedOnce = false
                  end
               end
            end
            check(folder:FindFirstChild("Dad"))
            check(folder:FindFirstChild("PossesedDad"))
         end)
      end
   end,
})

UserInputService.InputBegan:Connect(function(input, gp)
   if gp then return end
   if input.KeyCode == Enum.KeyCode.Q then
      SafeTeleport(EmergencyBedPos)
      Rayfield:Notify({Title = "Emergency", Content = "Bedroom!", Duration = 2})
   end
   if input.KeyCode == Enum.KeyCode.E then
      if GrabWaterStep == 0 then
         SafeTeleport(WaterPos2)
         GrabWaterStep = 1
      elseif GrabWaterStep == 1 then
         GrabWaterStep = 0
         Rayfield:Notify({Title = "Grab Water", Content = "Done", Duration = 2})
      end
   end
end)

-- PLAYER TAB
PlayerTab:CreateSection("ESP")

local function UpdateDadESP()
   local folder = GetDadFolder()
   if not folder then return end
   local dad = folder:FindFirstChild("Dad")
   if dad and (not DadHighlight or DadHighlight.Parent ~= dad) then
      if DadHighlight then DadHighlight:Destroy() end
      local h = Instance.new("Highlight")
      h.FillColor = Color3.fromRGB(255, 50, 50)
      h.OutlineColor = Color3.fromRGB(255, 255, 0)
      h.FillTransparency = 0.45
      h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
      h.Parent = dad
      DadHighlight = h
   end
   local poss = folder:FindFirstChild("PossesedDad")
   if poss and (not PossessedHighlight or PossessedHighlight.Parent ~= poss) then
      if PossessedHighlight then PossessedHighlight:Destroy() end
      local h = Instance.new("Highlight")
      h.FillColor = Color3.fromRGB(180, 0, 255)
      h.OutlineColor = Color3.fromRGB(255, 255, 0)
      h.FillTransparency = 0.45
      h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
      h.Parent = poss
      PossessedHighlight = h
   end
end

PlayerTab:CreateToggle({
   Name = "ESP Dad + Possessed Dad",
   CurrentValue = false,
   Flag = "ESPDad",
   Callback = function(Value)
      ESPDadEnabled = Value
      if Value then
         task.spawn(function()
            while ESPDadEnabled do
               UpdateDadESP()
               task.wait(0.8)
            end
            if DadHighlight then DadHighlight:Destroy() DadHighlight = nil end
            if PossessedHighlight then PossessedHighlight:Destroy() PossessedHighlight = nil end
         end)
      end
   end,
})
